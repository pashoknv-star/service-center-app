Обновлённая модель данных

4.1. Основные типы

type StatusCode =
  | 'vacation'
  | 'sick'
  | 'trip'
  | 'day_off'
  | 'work';

type ApprovalStatus =
  | 'draft'
  | 'pending'
  | 'approved'
  | 'rejected';

interface Group {
  id: string;
  name: string;
  sortOrder: number;
}

interface Position {
  id: string;
  name: string;
  sortOrder: number;
}

interface Employee {
  id: string;
  fullName: string;
  positionId: string;
  location: string;
  groupId: string;
  phone: string;
  photoUrl: string;
  isTechnical: boolean;
  requiresFullProfile: boolean;
  sortOrder: number;
  createdAt: string;
  updatedAt: string;
}

interface ScheduleEntry {
  employeeId: string;
  date: string; // YYYY-MM-DD
  status: StatusCode;
  comment?: string;
  approvalStatus: ApprovalStatus;
  updatedAt: string;
}

interface HolidayDay {
  date: string; // YYYY-MM-DD
  type: 'weekend' | 'holiday' | 'transferred_day_off' | 'working_weekend';
  label?: string;
}

interface VacationBalance {
  employeeId: string;
  year: number;
  totalDays: number;
  usedDays: number;
  remainingDays: number;
}

interface User {
  id: string;
  username: string;
  displayName: string;
  canEdit: boolean;
  isAdmin: boolean;
}

interface ScheduleState {
  version: number;
  year: number;
  groups: Group[];
  positions: Position[];
  employees: Employee[];
  entries: Record<string, ScheduleEntry>; // key = employeeId:date
  holidays: Record<string, HolidayDay>; // key = date
  vacationBalances: Record<string, VacationBalance>; // key = employeeId:year
  updatedAt: string;
  savedAt?: string;
}

4.2. Ключ записи статуса
const entryKey = `${employeeId}:${date}`;

Пример:
employee_001:2026-07-15

4.3. Ключ баланса отпуска
const balanceKey = `${employeeId}:${year}`;

Пример:
1. employee_001:2026
